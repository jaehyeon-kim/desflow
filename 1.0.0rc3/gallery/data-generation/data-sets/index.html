    Data Sets

    Seven data sets, each generated by one blueprint: a retail set written to three sinks, a film rental store, auction streams, a group-by table, taxi rides, labelled text and a point-in-time training set. Each entry is a complete file in three forms, so you can run it, read how it is built and change it.

    Before You Start

    The commands run from the root of a clone of the repository, where the files are in examples/gallery/:

    1
2
    git clone https://github.com/jaehyeon-kim/desflow.git
cd desflow

    An entry that needs a service or an extra says so. Services for the Examples shows how to start the services. A file with a fixed random_seed and factor: 0 prints the same records on every run, and each output block below holds the first events of the YAML file. The Builder and Low-level files publish the same records.

# two seconds. Each order picks a customer and a product at random and is followed by
# its invoice. The reference data goes to PostgreSQL tables, the orders to a Kafka
# topic and the invoices to a Parquet file, each sink taking the records of its own

#
# Needs: a broker and a database, with odctl up kafka-lite and odctl up postgres, and

# Run it with: desflow run examples/gallery/retail_records_to_three_sinks.yaml

simulation:
  sim_id: Retail
  factor: 0
  random_seed: 42
  logical_start_time: 2026-01-01T00:00:00

egress:
  - type: Postgres
    when: {expr: "record.task == 'customer'"}
    config:
      connection_dsn: postgresql://user:password@127.0.0.1:5432/odctl
      tables:
        customers:
          columns: {customer_id: INT, name: TEXT, country: TEXT}
          primary_key: customer_id
  - type: Postgres
    when: {expr: "record.task == 'product'"}
    config:
      connection_dsn: postgresql://user:password@127.0.0.1:5432/odctl
      tables:
        products:
          columns: {product_id: INT, title: TEXT, price: REAL}
          primary_key: product_id
  - type: Kafka
    when: {expr: "record.task == 'order'"}
    config:
      bootstrap_servers: ${KAFKA_BOOTSTRAP_SERVERS:-localhost:9092}
      event_topic: retail-orders
  - type: Parquet
    when: {expr: "record.task == 'invoice'"}
    config: {default_path: "${DEST_PATH:-data}/invoices.parquet"}

arrivals:
  customers: {initial: 20, spawn: customer}
  products: {initial: 10, spawn: product}
  orders: {dist: exponential, rate: 0.5, spawn: order}

tasks:
  customer:
    payload:
      customer_id: {gen: sequence, start: 1}
      name: {gen: fake, method: name}
      country:
        gen: weighted
        options: {AU: 5, US: 3, GB: 2}

  product:
    payload:
      product_id: {gen: sequence, start: 1}
      title:
        gen: one_of
        choices: [Desk lamp, Notebook, Water bottle, Backpack, Headphones, Keyboard]
      price: {gen: uniform, low: 2, high: 200, decimals: 2}

  order:
    vars:
      product: {gen: lookup, task: product}
      price: {gen: var, name: product, path: price}
      quantity: {gen: integer, bounds: [1, 5]}
    payload:
      order_id: {gen: sequence, start: 1}
      customer_id: {gen: lookup, task: customer, path: customer_id}
      product_id: {gen: var, name: product, path: product_id}
      quantity: {gen: var, name: quantity}
      total: {gen: expr, expr: "round(price * quantity, 2)"}
    next: invoice

  invoice:
    payload:
      order_id: {gen: upstream, path: order_id}
      amount: {gen: upstream, path: total}

run: {until: 10 min}

    """Retail records go to PostgreSQL, Kafka and Parquet, builder API.

The builder version of `retail_records_to_three_sinks.yaml`: 20 customers and 10
products at the start, an order about every two seconds and an invoice for each order.
The customers and products go to PostgreSQL tables, the orders to a Kafka topic and the
invoices to a Parquet file.

Needs: a broker and a database, with odctl up kafka-lite and odctl up postgres, and the
kafka, postgres, parquet and fake extras. KAFKA_BOOTSTRAP_SERVERS sets the broker
address, and DEST_PATH the folder of the Parquet file (data by default).
Run it with: python examples/gallery/retail_records_to_three_sinks_builder.py
"""

import logging
import os
from datetime import datetime, timezone

from desflow import (
    KafkaEgress,
    ParquetStorageEgress,
    PostgresEgress,
    SimulationContext,
)
from desflow.values import Template

UNTIL = 600
DSN = "postgresql://user:password@127.0.0.1:5432/odctl"


def build() -> SimulationContext:
    """Builds the shop, with a sink for each kind of record."""
    app = SimulationContext(
        sim_id="Retail",
        factor=0,
        random_seed=42,
        logical_start_time=datetime(2026, 1, 1, tzinfo=timezone.utc),
    )
    app.add_egress(
        PostgresEgress(
            DSN,
            tables={
                "customers": {
                    "columns": {
                        "customer_id": "INT",
                        "name": "TEXT",
                        "country": "TEXT",
                    },
                    "primary_key": "customer_id",
                }
            },
        ),
        when=lambda record: record.get("task") == "customer",
    )
    app.add_egress(
        PostgresEgress(
            DSN,
            tables={
                "products": {
                    "columns": {"product_id": "INT", "title": "TEXT", "price": "REAL"},
                    "primary_key": "product_id",
                }
            },
        ),
        when=lambda record: record.get("task") == "product",
    )
    app.add_egress(
        KafkaEgress(
            bootstrap_servers=os.environ.get(
                "KAFKA_BOOTSTRAP_SERVERS", "localhost:9092"
            ),
            event_topic="retail-orders",
        ),
        when=lambda record: record.get("task") == "order",
    )
    app.add_egress(
        ParquetStorageEgress(
            default_path=f"{os.environ.get('DEST_PATH', 'data')}/invoices.parquet"
        ),
        when=lambda record: record.get("task") == "invoice",
    )
    app.add_initial_arrival("customers", 20, spawn="customer")
    app.add_initial_arrival("products", 10, spawn="product")
    app.add_arrival("orders", "exponential", rate=0.5, spawn="order")
    customer_payload = Template(
        {
            "customer_id": {"gen": "sequence", "start": 1},
            "name": {"gen": "fake", "method": "name"},
            "country": {"gen": "weighted", "options": {"AU": 5, "US": 3, "GB": 2}},
        },
        name="tasks.customer",
    )
    product_payload = Template(
        {
            "product_id": {"gen": "sequence", "start": 1},
            "title": {
                "gen": "one_of",
                "choices": [
                    "Desk lamp",
                    "Notebook",
                    "Water bottle",
                    "Backpack",
                    "Headphones",
                    "Keyboard",
                ],
            },
            "price": {"gen": "uniform", "low": 2, "high": 200, "decimals": 2},
        },
        name="tasks.product",
    )
    order_payload = Template(
        {
            "order_id": {"gen": "sequence", "start": 1},
            "customer_id": {"gen": "lookup", "task": "customer", "path": "customer_id"},
            "product_id": {"gen": "var", "name": "product", "path": "product_id"},
            "quantity": {"gen": "var", "name": "quantity"},
            "total": {"gen": "expr", "expr": "round(price * quantity, 2)"},
        },
        name="tasks.order",
        vars={
            "product": {"gen": "lookup", "task": "product"},
            "price": {"gen": "var", "name": "product", "path": "price"},
            "quantity": {"gen": "integer", "bounds": [1, 5]},
        },
    )
    invoice_payload = Template(
        {
            "order_id": {"gen": "upstream", "path": "order_id"},
            "amount": {"gen": "upstream", "path": "total"},
        },
        name="tasks.invoice",
    )

    @app.task(name="customer", history=1_000_000)
    def customer(task_id):
        """Gives one customer."""
        return customer_payload

    @app.task(name="product", history=1_000_000)
    def product(task_id):
        """Gives one product."""
        return product_payload

    @app.task(name="order", next="invoice")
    def order(task_id):
        """Takes one order."""
        return order_payload

    @app.task(name="invoice")
    def invoice(task_id):
        """Bills the order."""
        return invoice_payload

    return app


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, format="%(levelname)s %(message)s")
    build().run(until=UNTIL)

    """Retail records go to PostgreSQL, Kafka and Parquet, low-level API.

The low-level version of `retail_records_to_three_sinks.yaml`: plain SimPy on a
`DynamicRealtimeEnvironment`, which publishes each record with `publish_event`. Each
draw is made with `Sampler.generator(<name>)`, from the random sequence the blueprint's
generator at that place uses, so the draws are the blueprint's. A name is drawn with a
Faker that is seeded from that sequence.

Needs: a broker and a database, with odctl up kafka-lite and odctl up postgres, and the
kafka, postgres, parquet and fake extras. KAFKA_BOOTSTRAP_SERVERS sets the broker
address, and DEST_PATH the folder of the Parquet file (data by default).
Run it with: python examples/gallery/retail_records_to_three_sinks_low_level.py
"""

import itertools
import logging
import os
from datetime import datetime, timezone

from faker import Faker

from desflow import (
    DistributionConfig,
    DynamicRealtimeEnvironment,
    KafkaEgress,
    ParquetStorageEgress,
    PostgresEgress,
    Sampler,
)

UNTIL = 600
START = datetime(2026, 1, 1, tzinfo=timezone.utc)
ENVIRONMENT = {"logical_start_time": START, "timezone": "UTC"}
DSN = "postgresql://user:password@127.0.0.1:5432/odctl"
COUNTRIES = {"AU": 5, "US": 3, "GB": 2}
TITLES = ["Desk lamp", "Notebook", "Water bottle", "Backpack", "Headphones", "Keyboard"]


def weighted(rng, options):
    """One key of `options`, chosen by its weight, as the `weighted` generator does."""
    point = rng.random() * sum(options.values())
    running = 0.0
    for value, weight in options.items():
        running += weight
        if point < running:
            return value
    return value


def build(env: DynamicRealtimeEnvironment) -> None:
    """Starts the arrivals on `env`."""
    sampler = Sampler(seed=42)
    faker = Faker("en_US")
    faker.seed_instance(
        int(sampler.generator("tasks.customer.payload.name#fake").integers(2**32))
    )
    country = sampler.generator("tasks.customer.payload.country")
    title = sampler.generator("tasks.product.payload.title")
    price = sampler.generator("tasks.product.payload.price")
    product_row = sampler.generator("tasks.order.vars.product")
    quantity = sampler.generator("tasks.order.vars.quantity")
    customer_row = sampler.generator("tasks.order.payload.customer_id")
    customers: list = []  # the ids of the customers so far, oldest first
    products: list = []  # the products so far, oldest first
    counts = {"customer": 0, "product": 0, "order": 0, "invoice": 0}

    def publish(key, task, payload):
        """Publishes a finished payload as the next instance of its task."""
        env.publish_event(key, payload, task=task, instance=counts[task])
        counts[task] += 1

    def customer():
        row = {
            "customer_id": len(customers) + 1,
            "name": faker.name(),
            "country": weighted(country, COUNTRIES),
        }
        customers.append(row["customer_id"])
        return row

    def product():
        row = {
            "product_id": len(products) + 1,
            "title": TITLES[int(title.integers(len(TITLES)))],
            "price": round(2 + (200 - 2) * price.random(), 2),
        }
        products.append(row)
        return row

    def order(number):
        """An order, and the invoice for it at once."""
        key = f"orders-{number}"
        chosen = products[int(product_row.integers(len(products)))]
        count = int(quantity.integers(1, 6))
        taken = {
            "order_id": number,
            "customer_id": customers[int(customer_row.integers(len(customers)))],
            "product_id": chosen["product_id"],
            "quantity": count,
            "total": round(chosen["price"] * count, 2),
        }
        publish(key, "order", taken)
        publish(key, "invoice", {"order_id": number, "amount": taken["total"]})

    gap = DistributionConfig(dist="exponential", rate=0.5)
    gap.stream = "arrival.orders"

    def source():
        for number in itertools.count(1):
            yield env.timeout(sampler.sample(gap))
            order(number)

    def start(name, task, make, number):
        yield env.timeout(0)
        publish(f"{name}-{number}", task, make())

    env.process(source())
    for number in range(1, 21):
        env.process(start("customers", "customer", customer, number))
    for number in range(1, 11):
        env.process(start("products", "product", product, number))


def run() -> None:
    """Runs the model unpaced, with a sink for each kind of record."""
    env = DynamicRealtimeEnvironment(factor=0, **ENVIRONMENT)
    bootstrap = os.environ.get("KAFKA_BOOTSTRAP_SERVERS", "localhost:9092")
    sinks = [
        (
            PostgresEgress(
                DSN,
                tables={
                    "customers": {
                        "columns": {
                            "customer_id": "INT",
                            "name": "TEXT",
                            "country": "TEXT",
                        },
                        "primary_key": "customer_id",
                    }
                },
            ),
            "customer",
        ),
        (
            PostgresEgress(
                DSN,
                tables={
                    "products": {
                        "columns": {
                            "product_id": "INT",
                            "title": "TEXT",

    """A film rental store: films, customers, rentals, payments and returns, low-level API.

The low-level version of `film_rental_store.yaml`: plain SimPy on a `DynamicRealtimeEnvironment`,
which publishes each record with `publish_event`. Each draw is made with
`Sampler.generator(<name>)`, from the random sequence the blueprint's generator at that
place uses, so the draws are the blueprint's. A name is drawn with a Faker that is
seeded from that sequence.

Needs: the fake extra.
Run it with: python examples/gallery/film_rental_store_low_level.py
"""

import itertools
import logging
from datetime import datetime, timedelta, timezone

from faker import Faker

from desflow import (
    ConsoleEgress,
    DistributionConfig,
    DynamicRealtimeEnvironment,
    Sampler,
)

UNTIL = 7 * 86400
START = datetime(2026, 1, 1, tzinfo=timezone.utc)
ENVIRONMENT = {"logical_start_time": START, "timezone": "UTC"}
ADJECTIVES = ["Silent", "Broken", "Golden", "Hidden", "Last"]
NOUNS = ["River", "Garden", "Highway", "Letter", "Harbour"]
RATINGS = {"G": 2, "PG": 3, "PG-13": 3, "R": 3, "NC-17": 1}
RATES = [0.99, 2.99, 4.99]


def weighted(rng, options):
    """One key of `options`, chosen by its weight, as the `weighted` generator does."""
    point = rng.random() * sum(options.values())
    running = 0.0
    for value, weight in options.items():
        running += weight
        if point < running:
            return value
    return value


def one_of(rng, choices):
    """One of `choices`, each equally likely, as the `one_of` generator does."""
    return choices[int(rng.integers(len(choices)))]


def build(env: DynamicRealtimeEnvironment) -> None:
    """Starts the arrivals on `env`."""
    sampler = Sampler(seed=42)

    def faker_of(place):
        faker = Faker("en_US")
        faker.seed_instance(int(sampler.generator(f"{place}#fake").integers(2**32)))
        return faker

    first_names = faker_of("tasks.customer.payload.first_name")
    last_names = faker_of("tasks.customer.payload.last_name")
    adjective = sampler.generator("tasks.film.payload.title.names.adjective")
    noun = sampler.generator("tasks.film.payload.title.names.noun")
    rating = sampler.generator("tasks.film.payload.rating")
    rate = sampler.generator("tasks.film.payload.rental_rate")
    length = sampler.generator("tasks.film.payload.length")
    active = sampler.generator("tasks.customer.payload.active")
    film_row = sampler.generator("tasks.rental.vars.film")
    customer_row = sampler.generator("tasks.rental.payload.customer_id")
    stay = sampler.generator("tasks.return.wait.timeout")
    films: list = []  # the films so far, oldest first
    customers: list = []  # the ids of the customers so far, oldest first
    counts = {"film": 0, "customer": 0, "rental": 0, "payment": 0, "return": 0}

    def iso():
        """The logical time now, as a timestamp text."""
        moment = START + timedelta(seconds=env.now)
        return moment.isoformat(timespec="milliseconds")

    def publish(key, task, payload, instance=None):
        """Publishes a finished payload, as the next instance of its task unless the
        task has queued one."""
        if instance is None:
            instance = counts[task]
            counts[task] += 1
        env.publish_event(key, payload, task=task, instance=instance)

    def film():
        row = {
            "film_id": len(films) + 1,
            "title": f"{one_of(adjective, ADJECTIVES)} {one_of(noun, NOUNS)}",
            "rating": weighted(rating, RATINGS),
            "rental_rate": one_of(rate, RATES),
            "length": int(length.integers(60, 181)),
        }
        films.append(row)
        return row

    def customer():
        row = {
            "customer_id": len(customers) + 1,
            "first_name": first_names.first_name(),
            "last_name": last_names.last_name(),
            "active": bool(active.random() < 0.9),
        }
        customers.append(row["customer_id"])
        return row

    def rental(number):
        """A rental, its payment at once, and its return after the stay."""
        key = f"rentals-{number}"
        chosen = films[int(film_row.integers(len(films)))]
        rented = {
            "rental_id": number,
            "rental_date": iso(),
            "film_id": chosen["film_id"],
            "rental_rate": chosen["rental_rate"],
            "customer_id": customers[int(customer_row.integers(len(customers)))],
        }
        publish(key, "rental", rented)
        paid = {
            "rental_id": rented["rental_id"],
            "customer_id": rented["customer_id"],
            "amount": rented["rental_rate"],
            "payment_date": iso(),
        }
        publish(key, "payment", paid)
        instance = counts["return"]  # the return waits, so it is queued first
        counts["return"] += 1
        env.publish_event(key, {"status": "queued"}, task="return", instance=instance)
        yield env.timeout(86400 + (604800 - 86400) * stay.random())  # 1 to 7 days
        env.publish_event(key, {"status": "started"}, task="return", instance=instance)
        returned = {"rental_id": rented["rental_id"], "return_date": iso()}
        publish(key, "return", returned, instance)

    gap = DistributionConfig(dist="exponential", rate=0.002)
    gap.stream = "arrival.rentals"

    def source():
        for number in itertools.count(1):
            yield env.timeout(sampler.sample(gap))
            env.process(rental(number))

    def start(name, task, make, number):
        yield env.timeout(0)
        publish(f"{name}-{number}", task, make())

    env.process(source())
    for number in range(1, 21):
        env.process(start("films", "film", film, number))
    for number in range(1, 16):
        env.process(start("customers", "customer", customer, number))


def run() -> None:
    """Runs the model unpaced into the console."""
    env = DynamicRealtimeEnvironment(factor=0, **ENVIRONMENT)
    env.setup_egress([ConsoleEgress()])
    build(env)
    try:
        env.run(until=UNTIL)
    finally:
        env.teardown()


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, format="%(levelname)s %(message)s")
    run()

    """Auction streams: people, auctions and bids, low-level API.

The low-level version of `auction_streams.yaml`: plain SimPy on a `DynamicRealtimeEnvironment`,
which publishes each record with `publish_event`. Each draw is made with
`Sampler.generator(<name>)`, from the random sequence the blueprint's generator at that
place uses, so the draws are the blueprint's. A name is drawn with a Faker that is
seeded from that sequence.

Needs: the fake extra.
Run it with: python examples/gallery/auction_streams_low_level.py
"""

import itertools
import logging
from datetime import datetime, timedelta, timezone

from faker import Faker

from desflow import (
    ConsoleEgress,
    DistributionConfig,
    DynamicRealtimeEnvironment,
    Sampler,
)

UNTIL = 600
START = datetime(2026, 1, 1, tzinfo=timezone.utc)
ENVIRONMENT = {"logical_start_time": START, "timezone": "UTC"}
CITIES = ["Phoenix", "Boise", "Cheyenne", "Portland", "Kent"]
STATES = {"AZ": 3, "ID": 2, "WY": 1, "OR": 3, "WA": 3}


def weighted(rng, options):
    """One key of `options`, chosen by its weight, as the `weighted` generator does."""
    point = rng.random() * sum(options.values())
    running = 0.0
    for value, weight in options.items():
        running += weight
        if point < running:
            return value
    return value


def integer(rng, low, high):
    """A whole number from `low` to `high`, both included, as `integer` draws it."""
    return int(rng.integers(low, high + 1))


def build(env: DynamicRealtimeEnvironment) -> None:
    """Starts the arrivals on `env`."""
    sampler = Sampler(seed=42)
    faker = Faker("en_US")
    faker.seed_instance(
        int(sampler.generator("tasks.person.payload.name#fake").integers(2**32))
    )
    city = sampler.generator("tasks.person.payload.city")
    state = sampler.generator("tasks.person.payload.state")
    seller = sampler.generator("tasks.auction.payload.seller")
    category = sampler.generator("tasks.auction.payload.category")
    reserve = sampler.generator("tasks.auction.payload.reserve")
    auction = sampler.generator("tasks.bid.payload.auction")
    bidder = sampler.generator("tasks.bid.payload.bidder")
    price = sampler.generator("tasks.bid.payload.price")
    people: list = []  # the ids of the people so far, oldest first
    auctions: list = []  # the ids of the auctions so far, oldest first
    counts = {"person": 0, "auction": 0, "bid": 0}

    def iso(offset=0):
        """The logical time now plus `offset` seconds, as a timestamp text."""
        moment = START + timedelta(seconds=env.now + offset)
        return moment.isoformat(timespec="milliseconds")

    def person():
        counts["person"] += 1
        row = {
            "person_id": 999 + counts["person"],
            "name": faker.name(),
            "city": CITIES[int(city.integers(len(CITIES)))],
            "state": weighted(state, STATES),
        }
        people.append(row["person_id"])
        return row

    def new_auction():
        counts["auction"] += 1
        row = {
            "auction_id": counts["auction"],
            "seller": people[int(seller.integers(len(people)))],
            "category": integer(category, 1, 10),
            "reserve": integer(reserve, 100, 1000),
            "expires": iso(3600),
        }
        auctions.append(row["auction_id"])
        return row

    def bid():
        counts["bid"] += 1
        return {
            "auction": auctions[int(auction.integers(len(auctions)))],
            "bidder": people[int(bidder.integers(len(people)))],
            "price": integer(price, 100, 2000),
            "bid_time": iso(),
        }

    def arrival(name, task, make, rate, initial=0):
        """Starts `initial` entities at time 0, then one after each gap."""
        gap = DistributionConfig(dist="exponential", rate=rate)
        gap.stream = f"arrival.{name}"

        def start(number):
            key = f"{name}-{number}"
            payload = make()
            env.publish_event(key, payload, task=task, instance=counts[task] - 1)

        def source():
            for number in itertools.count(initial + 1):
                yield env.timeout(sampler.sample(gap))
                start(number)

        def first(number):
            yield env.timeout(0)
            start(number)

        env.process(source())
        for number in range(1, initial + 1):
            env.process(first(number))

    arrival("people", "person", person, 0.02, initial=10)
    arrival("auctions", "auction", new_auction, 0.06, initial=5)
    arrival("bids", "bid", bid, 0.92)


def run() -> None:
    """Runs the model unpaced into the console."""
    env = DynamicRealtimeEnvironment(factor=0, **ENVIRONMENT)
    env.setup_egress([ConsoleEgress()])
    build(env)
    try:
        env.run(until=UNTIL)
    finally:
        env.teardown()


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, format="%(levelname)s %(message)s")
    run()

    """Taxi rides with a pickup, a drop-off, a distance and a fare, low-level API.

The low-level version of `taxi_rides.yaml`: plain SimPy on a `DynamicRealtimeEnvironment`,
which publishes each record with `publish_event`. Each draw is made with
`Sampler.generator(<name>)`, from the random sequence the blueprint's generator at that
place uses, so the draws are the blueprint's.

Needs: nothing
Run it with: python examples/gallery/taxi_rides_low_level.py
"""

import itertools
import logging
from datetime import datetime, timedelta, timezone

from desflow import (
    ConsoleEgress,
    DistributionConfig,
    DynamicRealtimeEnvironment,
    Sampler,
)

UNTIL = 600
START = datetime(2026, 1, 1, tzinfo=timezone.utc)
ENVIRONMENT = {"logical_start_time": START, "timezone": "UTC"}
PASSENGERS = {1: 70, 2: 15, 3: 5, 4: 5, 5: 5}
PAYMENT = {"card": 65, "cash": 33, "dispute": 2}


def weighted(rng, options):
    """One key of `options`, chosen by its weight, as the `weighted` generator does."""
    point = rng.random() * sum(options.values())
    running = 0.0
    for value, weight in options.items():
        running += weight
        if point < running:
            return value
    return value


def uniform(rng, low, high, decimals):
    """A number in [low, high), as the `uniform` generator draws it."""
    return round(low + (high - low) * rng.random(), decimals)


def build(env: DynamicRealtimeEnvironment) -> None:
    """Starts the arrivals on `env`."""
    sampler = Sampler(seed=42)
    place = "tasks.ride"
    passengers = sampler.generator(f"{place}.payload.passengers")
    latitude = sampler.generator(f"{place}.payload.pickup_latitude")
    longitude = sampler.generator(f"{place}.payload.pickup_longitude")
    payment = sampler.generator(f"{place}.payload.payment")
    distance = sampler.generator(f"{place}.vars.distance")
    minutes = sampler.generator(f"{place}.vars.minutes")
    gap = DistributionConfig(dist="exponential", rate=0.1)
    gap.stream = "arrival.rides"

    def source():
        for number in itertools.count(1):
            yield env.timeout(sampler.sample(gap))
            km = uniform(distance, 0.5, 20, 2)
            length = uniform(minutes, 3, 40, 1)
            pickup = START + timedelta(seconds=env.now)
            dropoff = datetime.fromisoformat(
                pickup.isoformat(timespec="milliseconds")
            ) + timedelta(seconds=length * 60)
            ride = {
                "ride_id": number,
                "pickup_time": pickup.isoformat(timespec="milliseconds"),
                "dropoff_time": dropoff.isoformat(timespec="milliseconds"),
                "passengers": weighted(passengers, PASSENGERS),
                "pickup_latitude": uniform(latitude, 40.7, 40.8, 5),
                "pickup_longitude": uniform(longitude, -74.02, -73.93, 5),
                "distance_km": km,
                "fare": round(3 + 2.2 * km + 0.4 * length, 2),
                "payment": weighted(payment, PAYMENT),
            }
            env.publish_event(f"rides-{number}", ride, task="ride", instance=number - 1)

    env.process(source())


def run() -> None:
    """Runs the model unpaced into the console."""
    env = DynamicRealtimeEnvironment(factor=0, **ENVIRONMENT)
    env.setup_egress([ConsoleEgress()])
    build(env)
    try:
        env.run(until=UNTIL)
    finally:
        env.teardown()


if __name__ == "__main__":

    run()

        ("Please refund order {n}.", 1),
    ],
    "cancel": [
        ("Cancel my order, I changed my mind.", 1),
        ("Please stop my subscription.", 1),
    ],
    "hello": [("Hi, I have a question.", 1), ("Hello, is anyone there?", 1)],
}
# The texts of a review, by sentiment. A text takes the product
REVIEWS = {
    "positive": [
        ("The {product} is great and arrived early.", 1),
        ("I love this {product}, I would buy it again.", 1),
    ],
    "neutral": [("The {product} is fine, nothing special.", 1)],
    "negative": [
        ("The {product} broke after a week.", 1),
        ("I would not recommend this {product}.", 1),
    ],
}


def position(rng, weights):
    """The position of one weight, chosen by its size, as the `weighted` generator does."""
    point = rng.random() * sum(weights)
    running = 0.0
    for index, weight in enumerate(weights):
        running += weight
        if point < running:
            return index
    return len(weights) - 1


def build(env: DynamicRealtimeEnvironment) -> None:
    """Starts the arrivals on `env`."""
    sampler = Sampler(seed=42)
    intent = sampler.generator("tasks.message.vars.intent")
    sentiment = sampler.generator("tasks.review.vars.sentiment")
    product = sampler.generator("tasks.review.vars.product")
    counts = {"message": 0, "review": 0}

    def message_text(label):
        """The text of a message: a template chosen by weight, with an order number."""
        place = f"tasks.message.payload.text.cases.{label}"
        templates = MESSAGES[label]
        chosen = position(sampler.generator(place), [w for _, w in templates])
        template = templates[chosen][0]
        number = sampler.generator(f"{place}.choices.{chosen}.value.names.n")
        return template.format(n=int(number.integers(1000, 10000)))

    def review_text(label, name):
        """The text of a review: a template chosen by weight, for the product."""
        templates = REVIEWS[label]
        weights = [w for _, w in templates]
        place = f"tasks.review.payload.text.cases.{label}"
        chosen = 0 if len(weights) == 1 else position(sampler.generator(place), weights)
        return templates[chosen][0].format(product=name)

    def message():
        counts["message"] += 1
        label = list(INTENTS)[position(intent, list(INTENTS.values()))]
        return {
            "message_id": counts["message"],
            "intent": label,
            "text": message_text(label),
        }

    def review():
        counts["review"] += 1
        label = list(SENTIMENTS)[position(sentiment, list(SENTIMENTS.values()))]
        name = PRODUCTS[int(product.integers(len(PRODUCTS)))]
        return {
            "review_id": counts["review"],
            "sentiment": label,
            "text": review_text(label, name),
        }

    def arrival(name, task, make, rate):
        gap = DistributionConfig(dist="exponential", rate=rate)
        gap.stream = f"arrival.{name}"

        def source():
            for number in itertools.count(1):
                yield env.timeout(sampler.sample(gap))
                env.publish_event(
                    f"{name}-{number}", make(), task=task, instance=number - 1
                )

        env.process(source())

    arrival("messages", "message", message, 0.5)
    arrival("reviews", "review", review, 0.2)


def run() -> None:
    """Runs the model unpaced into the console."""
    env = DynamicRealtimeEnvironment(factor=0, **ENVIRONMENT)
    env.setup_egress([ConsoleEgress()])
    build(env)
    try:
        env.run(until=UNTIL)
    finally:
        env.teardown()


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, format="%(levelname)s %(message)s")
    run()

    """A point-in-time training set: feature records and the label that follows each,
builder API.

The builder version of `point_in_time_training_set.yaml`: five customers, each with a
feature record every day and, 7 days after each, a label record.

Needs: nothing
Run it with: python examples/gallery/point_in_time_training_set_builder.py
"""

import logging
from datetime import datetime, timezone

from desflow import ConsoleEgress, SimulationContext
from desflow.values import Template

UNTIL = 40 * 86400
CUSTOMERS = 5


def build() -> SimulationContext:
    """Builds the customers. A blueprint's `count: 5` is five arrivals here, each with
    the key of its customer."""
    app = SimulationContext(
        sim_id="Churn",
        factor=0,
        random_seed=42,
        logical_start_time=datetime(2026, 1, 1, tzinfo=timezone.utc),
    )
    app.add_egress(ConsoleEgress())
    for number in range(1, CUSTOMERS + 1):
        app.add_arrival(
            f"customers{number}",
            "constant",
            value=86400,  # 1 day
            spawn="features",
            key=lambda n, number=number: f"customer-{number}",
        )
    features_payload = Template(
        {
            "customer_id": {"gen": "entity"},
            "as_of": {"gen": "now"},
            "sessions_30d": {"gen": "integer", "bounds": [0, 30]},
            "spend_30d": {"gen": "uniform", "low": 0, "high": 500, "decimals": 2},
        },
        name="tasks.features",
    )
    label_payload = Template(
        {
            "customer_id": {"gen": "entity"},
            "as_of": {"gen": "upstream", "path": "as_of"},
            "churned": {
                "gen": "when",
                "condition": "upstream.sessions_30d < 5",
                "then": {"gen": "boolean", "p": 0.6},
                "else": {"gen": "boolean", "p": 0.05},
            },
        },
        name="tasks.label",
    )

    @app.task(name="features", next="label")
    def features(task_id):
        """Gives the features of a customer."""
        return features_payload

    @app.task(name="label", wait={"timeout": "7 days"})
    def label(task_id):
        """Says, 7 days later, whether the customer left."""
        return label_payload

    return app


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, format="%(levelname)s %(message)s")
    build().run(until=UNTIL)

    """A point-in-time training set: feature records and the label that follows each, low-level API.

The low-level version of `point_in_time_training_set.yaml`: plain SimPy on a `DynamicRealtimeEnvironment`,
which publishes each record with `publish_event`. Each draw is made with
`Sampler.generator(<name>)`, from the random sequence the blueprint's generator at that
place uses, so the draws are the blueprint's.

Needs: nothing
Run it with: python examples/gallery/point_in_time_training_set_low_level.py
"""

import itertools
import logging
from datetime import datetime, timedelta, timezone

from desflow import ConsoleEgress, DynamicRealtimeEnvironment, Sampler

UNTIL = 40 * 86400
START = datetime(2026, 1, 1, tzinfo=timezone.utc)
ENVIRONMENT = {"logical_start_time": START, "timezone": "UTC"}


def build(env: DynamicRealtimeEnvironment) -> None:
    """Starts the customers on `env`."""
    sampler = Sampler(seed=42)
    sessions = sampler.generator("tasks.features.payload.sessions_30d")
    spend = sampler.generator("tasks.features.payload.spend_30d")
    few = sampler.generator("tasks.label.payload.churned.then")
    many = sampler.generator("tasks.label.payload.churned.else")
    counts = {"features": 0, "label": 0}

    def customer(key):
        for _ in itertools.count():
            yield env.timeout(86400)
            env.process(flow(key))

    def flow(key):
        """The features of a customer now, then the label 7 days later."""
        moment = START + timedelta(seconds=env.now)
        features = {
            "customer_id": key,
            "as_of": moment.isoformat(timespec="milliseconds"),
            "sessions_30d": int(sessions.integers(0, 31)),
            "spend_30d": round(500 * spend.random(), 2),
        }
        env.publish_event(key, features, task="features", instance=counts["features"])
        counts["features"] += 1
        instance = counts["label"]  # the label waits, so it is queued first
        counts["label"] += 1
        env.publish_event(key, {"status": "queued"}, task="label", instance=instance)
        yield env.timeout(7 * 86400)
        env.publish_event(key, {"status": "started"}, task="label", instance=instance)
        # A customer with few sessions leaves with a high chance
        chance, rng = (0.6, few) if features["sessions_30d"] < 5 else (0.05, many)
        label = {
            "customer_id": key,
            "as_of": features["as_of"],
            "churned": bool(rng.random() < chance),
        }
        env.publish_event(key, label, task="label", instance=instance)

    for number in range(1, 6):
        env.process(customer(f"customer-{number}"))


def run() -> None:
    """Runs the model unpaced into the console."""
    env = DynamicRealtimeEnvironment(factor=0, **ENVIRONMENT)
    env.setup_egress([ConsoleEgress()])
    build(env)
    try:
        env.run(until=UNTIL)
    finally:
        env.teardown()


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, format="%(levelname)s %(message)s")
    run()
