"""Shop model, low-level API. Prints each order_packed record as a JSON line."""

from desflow import (
    CapacityConfig,
    ConsoleEgress,
    DistributionConfig,
    DynamicRealtimeEnvironment,
    DynamicResource,
    SimParameter,
)

PARAMS = SimParameter(
    sim_id="Shop",
    arrival={"orders": DistributionConfig(dist="constant", value=2.0)},
    service={"packing": DistributionConfig(dist="constant", value=1.5)},
    resources={"packers": CapacityConfig(current_cap=1)},
)


def packed_only(record):
    value = record.get("value")
    return isinstance(value, dict) and value.get("event_type") == "order_packed"


def pack_order(env, packers, key, task_id):
    path_id = "Shop.service.packing"

    def publish(value):
        env.publish_event(key, value, task="pack_order", instance=task_id)

    publish({"path_id": path_id, "status": "queued"})
    with packers.request() as req:
        yield req
        publish({"path_id": path_id, "status": "started"})
        yield env.timeout(1.5)
        publish({"event_type": "order_packed"})


def arrivals(env, packers):
    yield env.timeout(2)
    n = 0
    while True:
        n += 1
        env.process(pack_order(env, packers, f"orders-{n}", n - 1))
        yield env.timeout(2)


def run():
    env = DynamicRealtimeEnvironment(factor=0, strict=False)
    env.registry.register_sim_parameter(PARAMS)
    env.setup_egress([ConsoleEgress(format="json")], predicates=[packed_only])
    packers = DynamicResource(env, "Shop", "packers")
    env.process(arrivals(env, packers))
    try:
        env.run(until=20)
    finally:
        env.teardown()


if __name__ == "__main__":
    run()
