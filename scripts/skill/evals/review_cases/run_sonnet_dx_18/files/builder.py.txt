"""Shop model, builder API. Prints each order_packed record as a JSON line."""

from desflow import ConsoleEgress, SimulationContext

app = (
    SimulationContext(sim_id="Shop", factor=0, random_seed=42)
    .add_egress(
        ConsoleEgress(format="json"),
        when=lambda record: record["value"].get("event_type") == "order_packed"
        if isinstance(record.get("value"), dict)
        else False,
    )
    .add_resource("packers", current_cap=1)
    .add_service("packing", dist="constant", value=1.5)
    .add_arrival("orders", gap=2, first_at=2)
)


@app.task(service_id="packing", resource_id="packers")
def pack_order(task_id: int, context):
    return {"event_type": "order_packed"}


@app.arrival_loop("orders")
def arrivals(context):
    task_id = 0
    while True:
        entity = yield context.wait_for_arrival("orders")
        context.spawn(pack_order(task_id, context, entity=entity))
        task_id += 1


if __name__ == "__main__":
    app.run(until=20)
