import json
import subprocess
import sys
from collections import Counter
from pathlib import Path

import pytest

HERE = Path(__file__).parent


def run_script(name):
    out = subprocess.run(
        [sys.executable, name], cwd=HERE, capture_output=True, text=True, check=True
    ).stdout
    return [json.loads(line) for line in out.splitlines() if line.strip()]


def run_yaml():
    out = subprocess.run(
        [str(Path(sys.executable).parent / "desflow"), "run", "model.yaml"],
        cwd=HERE,
        capture_output=True,
        text=True,
        check=True,
    ).stdout
    return [json.loads(line) for line in out.splitlines() if line.strip()]


@pytest.fixture(scope="module")
def records():
    return {
        "builder": run_script("builder.py"),
        "low_level": run_script("low_level.py"),
        "yaml": run_yaml(),
    }


def test_forms_give_same_records(records):
    assert records["builder"] == records["yaml"]
    assert records["low_level"] == records["yaml"]


@pytest.mark.parametrize("form", ["yaml", "builder", "low_level"])
def test_records(records, form):
    recs = records[form]
    assert recs and all(r["task"] == "member" for r in recs)
    assert all(r["value"] == {"event_type": "member"} for r in recs)
    by_time = Counter(r["sim_ts"] for r in recs)
    # firings at 5, 10, ..., 200 only; none after 200
    assert all(t % 5 == 0 and 5 <= t <= 200 for t in by_time)
    # group sizes vary, and some firings start nobody (size 0)
    assert len(set(by_time.values())) > 1
    assert len(by_time) < 40
    # keys are unique, numbered in order
    assert [r["key"] for r in recs] == [f"a-{i}" for i in range(1, len(recs) + 1)]
    # about 40 * 2 = 80 members expected; stay within a loose band
    assert 40 <= len(recs) <= 120
    assert sum(by_time.values()) == len(recs)
