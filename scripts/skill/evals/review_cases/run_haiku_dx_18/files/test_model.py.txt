"""Tests for the Shop model in model.yaml, builder.py and low_level.py.

Run from this folder with: pytest test_model.py
"""

import json
import os
import subprocess
import sys
from pathlib import Path

import pytest
import yaml

HERE = Path(__file__).resolve().parent
PY_DIR = Path(sys.executable).resolve().parent
DESFLOW = PY_DIR / "desflow"

# The records the sink must print: one order_packed event per order, every 2 seconds
# from 3.5 seconds: the order that arrives at 2 seconds packs until 3.5, and so on, up
# to the order that arrives at 18 seconds and packs until 19.5. The run ends at 20.
EXPECTED = [
    {
        "stream_type": "event",
        "sim_ts": 3.5 + 2 * i,
        "timestamp": f"2026-01-01T00:00:{3 + 2 * i:02d}.500+00:00",
        "key": f"orders-{i + 1}",
        "task": "pack_order",
        "instance": i,
        "value": {"event_type": "order_packed"},
    }
    for i in range(9)
]


def run_stdout(args):
    """Runs a command in this folder and returns its standard output lines."""
    result = subprocess.run(
        args,
        cwd=HERE,
        capture_output=True,
        text=True,
        check=False,
    )
    assert result.returncode == 0, result.stderr
    return result.stdout.splitlines()


@pytest.fixture(scope="module")
def yaml_lines():
    return run_stdout([str(DESFLOW), "run", "model.yaml"])


@pytest.fixture(scope="module")
def builder_lines():
    return run_stdout([sys.executable, "builder.py"])


@pytest.fixture(scope="module")
def low_level_lines():
    return run_stdout([sys.executable, "low_level.py"])


def test_blueprint_keeps_seed_42_as_a_literal():
    blueprint = yaml.safe_load((HERE / "model.yaml").read_text())
    assert blueprint["simulation"]["random_seed"] == 42
    assert blueprint["simulation"]["sim_id"] == "Shop"
    assert blueprint["simulation"]["factor"] == 0
    assert blueprint["run"]["until"] == 20


def test_builder_and_low_level_keep_seed_42_as_a_literal():
    assert "random_seed=42" in (HERE / "builder.py").read_text()
    assert "default_rng(42)" in (HERE / "low_level.py").read_text()


def test_yaml_prints_the_expected_records(yaml_lines):
    assert [json.loads(line) for line in yaml_lines] == EXPECTED


def test_builder_prints_the_expected_records(builder_lines):
    assert [json.loads(line) for line in builder_lines] == EXPECTED


def test_low_level_prints_the_expected_records(low_level_lines):
    assert [json.loads(line) for line in low_level_lines] == EXPECTED


def test_three_forms_print_the_same_lines(yaml_lines, builder_lines, low_level_lines):
    assert yaml_lines == builder_lines == low_level_lines


def test_only_order_packed_events_reach_standard_output(yaml_lines):
    for line in yaml_lines:
        record = json.loads(line)
        assert record["stream_type"] == "event"
        assert record["value"]["event_type"] == "order_packed"
