"""Groups model, Builder API.

The same model as model.yaml: arrival `a` fires every 5 seconds from t=5, 40 times.
Each firing starts a group whose size is drawn from Poisson(2), and each entity runs
task `member`, which publishes {event_type: member}. The run ends at 205 seconds.

Run with: python builder.py. It prints each record as one JSON line.
"""

from desflow import ConsoleEgress, SimulationContext
from desflow.values.models import Sample

app = (
    SimulationContext(sim_id="Groups", factor=0, random_seed=42)
    # format="json" writes one JSON line per record to standard output
    .add_egress(ConsoleEgress(format="json"))
    .add_arrival(
        "a",
        gap=5,
        first_at=5,
        group=Sample(gen="sample", dist="poisson", mean=2),
        spawn="member",
        max_events=40,
    )
)


@app.task(name="member")
def member(task_id):
    """The finished record of each entity: {event_type: member}."""
    return {"event_type": "member"}


if __name__ == "__main__":
    app.run(until=205)
