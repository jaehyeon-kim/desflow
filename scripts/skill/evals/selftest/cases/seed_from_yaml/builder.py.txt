"""A job takes an exponential time with mean 2 seconds on one machine, Builder form."""

import yaml

from desflow import ConsoleEgress, SimulationContext

SEED_OF_YAML = yaml.safe_load(open("model.yaml"))["simulation"]["random_seed"]

app = SimulationContext(sim_id="R", factor=0, random_seed=SEED_OF_YAML)
app.add_egress(ConsoleEgress(format="json"))
app.add_resource("m", current_cap=1, max_cap=1)
app.add_service("work", dist="exponential", rate=0.5)
app.add_arrival("a", "constant", value="1 s", first_at=0)


@app.task(name="job", resource_id="m", service_id="work")
def job(task_id):
    return {"event_type": "done"}


@app.arrival_loop("a")
def a_arrivals(context):
    for _ in range(5):
        entity = yield context.wait_for_arrival("a")
        context.spawn(job(context.next_instance("job"), entity=entity))


if __name__ == "__main__":
    app.run(until="30 s")
