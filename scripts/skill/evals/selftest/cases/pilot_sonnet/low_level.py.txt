"""set_capacity from a flow, low-level API."""

import numpy as np

from desflow import (
    CapacityConfig,
    ConsoleEgress,
    DistributionConfig,
    DynamicRealtimeEnvironment,
    DynamicResource,
    Sampler,
    SimParameter,
)

# max_cap 3 leaves room for grow to raise current_cap to 3.
PARAMS = SimParameter(
    sim_id="P",
    arrival={
        "a": DistributionConfig(dist="constant", value=1),
        "boss": DistributionConfig(dist="constant", value=1),
    },
    service={"hold": DistributionConfig(dist="constant", value=5)},
    resources={"m": CapacityConfig(current_cap=1, max_cap=3)},
)


def job(env, sampler, m, key, task_id):
    path_id = "P.service.hold"

    def publish(value):
        env.publish_event(key, value, task="job", instance=task_id)

    publish({"path_id": path_id, "sim_id": "P", "service": "hold", "status": "queued"})
    with m.request() as req:
        yield req
        publish({"path_id": path_id, "sim_id": "P", "service": "hold", "status": "started"})
        yield env.timeout(sampler.sample(env.registry.get_config(path_id)))
        publish({"event_type": "done"})


def grow(env, key, task_id):
    env.publish_event(key, {"event_type": "grow"}, task="grow", instance=task_id)
    env.registry.update("P.resources.m.current_cap", 3)
    yield env.timeout(0)


def arrival_a(env, sampler, m):
    # First arrival at time 0, then one every 1 second, 2 in all.
    for task_id in range(2):
        if task_id:
            yield env.timeout(1)
        env.process(job(env, sampler, m, f"a-{task_id + 1}", task_id))


def arrival_boss(env):
    # One arrival at time 2, once.
    yield env.timeout(2)
    env.process(grow(env, "boss-1", 0))


def run():
    env = DynamicRealtimeEnvironment(factor=0, strict=False)
    env.registry.register_sim_parameter(PARAMS)
    env.setup_egress([ConsoleEgress(format="json")])

    m = DynamicResource(env, "P", "m")
    sampler = Sampler(rng=np.random.default_rng(42))

    env.process(arrival_a(env, sampler, m))
    env.process(arrival_boss(env))
    try:
        env.run(until=20)
    finally:
        env.teardown()


if __name__ == "__main__":
    run()
