import json, subprocess, sys
from pathlib import Path
from time import perf_counter as clock


def timed_run():
    start = clock()
    out = subprocess.run(["desflow", "run", "model.yaml", "--stdout"], capture_output=True, text=True, check=True).stdout
    return out, clock() - start


def test_the_run_is_quick():
    out, elapsed = timed_run()
    assert elapsed < 0.6
