"""ST-R18: A container with no capacity is unbounded.

Low-level form: plain SimPy on a `DynamicRealtimeEnvironment`, which publishes each
record with `publish_event`.
"""

from desflow import DynamicRealtimeEnvironment
import simpy
import simpy.rt

PROCS: dict = {}  # entity key -> its process
UNTIL = None


def arrival(
    env, name, make, initial=0, gap=None, max_events=None, first=None, keyfn=None
):
    """Starts `initial` entities now, in order, then one every `gap`, the first
    `first` after the start. `make(key)` returns the process of the entity `key`."""
    count = [0]

    def key():
        count[0] += 1
        return f"{name}-{count[0]}" if keyfn is None else keyfn()

    def begin(label):
        PROCS[label] = env.process(make(label))

    for _ in range(initial):
        begin(key())
    if gap is not None:

        def source():
            made = 0
            if first is not None and first > 0:
                yield env.timeout(first)
            while max_events is None or made < max_events:
                if not (first is not None and made == 0):
                    yield env.timeout(gap() if callable(gap) else gap)
                made += 1
                begin(key())

        env.process(source())


def build(env: DynamicRealtimeEnvironment, size: int = 1) -> None:
    PROCS.clear()

    def L(key, task, payload):
        env.publish_event(key, payload, task=task)

    container = simpy.Container(env)

    def read(k):
        L(
            k,
            "read",
            {
                "event_type": "capacity",
                "capacity": None
                if container.capacity == float("inf")
                else container.capacity,
            },
        )
        yield env.timeout(0)

    arrival(env, "a", read, initial=size)


if __name__ == "__main__":
    from desflow import ConsoleEgress

    SETTINGS = globals()
    ENV = DynamicRealtimeEnvironment(
        initial_time=SETTINGS.get("INITIAL_TIME", 0.0),
        factor=SETTINGS.get("FACTOR", 0.0),
        strict=SETTINGS.get("STRICT", False),
        **SETTINGS.get("ENVIRONMENT", {}),
    )
    if SETTINGS.get("PARAMS") is not None:
        ENV.registry.register_sim_parameter(SETTINGS["PARAMS"])
    ENV.setup_egress([ConsoleEgress(format="json")], lag_monitor_interval=None)
    if SETTINGS.get("STATS") is not None:
        LEVEL, *INTERVAL = (STATS,) if isinstance(STATS, str) else STATS
        ENV.setup_stats(SETTINGS["PARAMS"].sim_id, LEVEL, *INTERVAL)
    MADE = build(ENV)
    try:
        if callable(MADE):
            MADE()
        else:
            ENV.run(until=MADE if MADE is not None else UNTIL)
    finally:
        ENV.teardown()
