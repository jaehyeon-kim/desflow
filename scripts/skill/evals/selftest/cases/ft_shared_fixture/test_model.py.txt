"""Tests of task P33."""

import json
import subprocess
import sys
from pathlib import Path

import pytest

HERE = Path(__file__).parent
FORMS = {
    "yaml": [str(Path(sys.executable).parent / "desflow"), "run", "model.yaml", "--stdout"],
    "builder": [sys.executable, "builder.py"],
    "low_level": [sys.executable, "low_level.py"],
}
ENGINE = ("queued", "started", "resumed")


def _events(command):
    done = subprocess.run(command, cwd=HERE, capture_output=True, text=True, check=True)
    rows = [json.loads(line) for line in done.stdout.splitlines() if line.startswith("{")]
    return [
        r
        for r in rows
        if r["stream_type"] == "event"
        and not (isinstance(r["value"], dict) and r["value"].get("status") in ENGINE)
    ]


@pytest.fixture(scope="module")
def all_events():
    """Runs each form once for the whole module."""
    return {form: _events(command) for form, command in FORMS.items()}


def _times(rows, task, key):
    return [r["sim_ts"] for r in rows if r["task"] == task and r["key"] == key]


@pytest.mark.parametrize("form", FORMS)
def test_the_first_job_is_done_after_5(all_events, form):
    assert _times(all_events[form], "job", "a-1") == [5]


@pytest.mark.parametrize("form", FORMS)
def test_the_second_job_starts_when_the_capacity_grows(all_events, form):
    assert _times(all_events[form], "grow", "boss-1") == [2]
    assert _times(all_events[form], "job", "a-2") == [7]
