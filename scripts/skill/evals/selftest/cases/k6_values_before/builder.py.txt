"""Task P33, Builder form."""

from desflow import ConsoleEgress, SimulationContext
from desflow.values import Template  # documented in the API reference

app = SimulationContext(sim_id="P", factor=0, random_seed=42)
app.add_egress(ConsoleEgress(format="json"))
app.add_resource("m", current_cap=1, max_cap=3)
app.add_service("work", dist="constant", value="5 s")
app.add_arrival("a", "constant", value="1 s", first_at=0)
app.add_arrival("boss", "constant", value="2 s")


@app.task(name="job", resource_id="m", service_id="work")
def job(task_id):
    return {"event_type": "done"}


@app.task(name="grow", on=[{"path": "P.resources.m.current_cap", "value": 3}])
def grow(task_id):
    return {"event_type": "grow"}


@app.arrival_loop("a")
def a_arrivals(context):
    for _ in range(2):
        entity = yield context.wait_for_arrival("a")
        context.spawn(job(context.next_instance("job"), entity=entity))


@app.arrival_loop("boss")
def boss_arrivals(context):
    entity = yield context.wait_for_arrival("boss")
    context.spawn(grow(context.next_instance("grow"), entity=entity))


if __name__ == "__main__":
    # The original runs until 20 seconds, well after the last job is done at 7
    app.run(until="20 s")
