"""Tests of P33 that run the Builder form in this process and the YAML form as a command."""

import json
import runpy
import subprocess
import sys
from pathlib import Path

HERE = Path(__file__).parent


def _rows(text):
    rows = []
    for line in text.splitlines():
        if line.startswith("{"):
            row = json.loads(line)
            if row["stream_type"] == "event" and "event_type" in row["value"]:
                rows.append((row["sim_ts"], row["key"], row["task"]))
    return rows


def builder_rows(capsys):
    capsys.readouterr()
    runpy.run_path(str(HERE / "builder.py"), run_name="__main__")
    return _rows(capsys.readouterr().out)


def yaml_rows():
    desflow = str(Path(sys.executable).parent / "desflow")
    done = subprocess.run([desflow, "run", "model.yaml", "--stdout"], cwd=HERE, capture_output=True, text=True, check=True)
    return _rows(done.stdout)


def test_the_builder_form_ends_the_second_job_at_7(capsys):
    assert (7.0, "a-2", "job") in builder_rows(capsys)


def test_the_yaml_form_ends_the_second_job_at_7():
    assert (7.0, "a-2", "job") in yaml_rows()
    assert (5.0, "a-1", "job") in yaml_rows()


def test_the_forms_give_the_same_records(capsys):
    assert builder_rows(capsys) == yaml_rows()
