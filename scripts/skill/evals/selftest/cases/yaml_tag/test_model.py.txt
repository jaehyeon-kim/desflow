"""Tests of task P33. Each form runs as its reader runs it."""

import json
import math
import subprocess
import sys
from pathlib import Path

import pytest

HERE = Path(__file__).parent


def records(command):
    done = subprocess.run(command, cwd=HERE, capture_output=True, text=True, check=True)
    rows = [json.loads(l) for l in done.stdout.splitlines() if l.startswith("{")]
    return [r for r in rows if "event_type" in r.get("value", {})]


FORMS = {
    "yaml": [
        str(Path(sys.executable).parent / "desflow"),
        "run",
        "model.yaml",
        "--sample",
        "1000",
    ],
    "builder": [sys.executable, "builder.py"],
    "low_level": [sys.executable, "low_level.py"],
}


@pytest.mark.parametrize("form", FORMS)
def test_the_second_job_starts_when_the_capacity_grows(form):
    rows = {(r["task"], r["key"]): r["sim_ts"] for r in records(FORMS[form])}
    assert rows[("grow", "boss-1")] == 2
    assert rows[("job", "a-1")] == 5
    # a-2 waits from 1, starts at 2 when the capacity is 3, and is done at 7
    assert rows[("job", "a-2")] == 7


@pytest.mark.parametrize("form", FORMS)
def test_each_job_carries_pi(form):
    done = [r for r in records(FORMS[form]) if r["value"]["event_type"] == "done"]
    assert len(done) == 2
    assert all(r["value"]["n"] == math.pi for r in done)
