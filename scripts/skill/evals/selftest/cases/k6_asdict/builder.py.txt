"""Builder API form of model.yaml. Prints each record as one JSON line."""

from desflow import ConsoleEgress, SimulationContext

app = (
    SimulationContext(sim_id="P", factor=0, random_seed=42)
    .add_egress(ConsoleEgress(format="json"))
    # max_cap is 3 so that grow can raise current_cap to 3
    .add_resource("m", current_cap=1, max_cap=3)
    .add_service("hold", dist="constant", value=5)
    .add_arrival("a", dist="constant", value=1, first_at=0, max_events=2, spawn="job")
    .add_arrival("boss", dist="constant", value=1, first_at=2, max_events=1, spawn="grow")
)

@app.task(service_id="hold", resource_id="m")
def job(task_id):
    return {"event_type": "done"}


@app.task(on=[{"path": "P.resources.m.current_cap", "value": 3}])
def grow(task_id):
    return {"event_type": "grow"}


if __name__ == "__main__":
    app.run(until=20)
