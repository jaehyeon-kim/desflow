"""DS-A5: Turn one source off and on from a flow.

Builder form of `model.yaml`.
"""

from desflow import SimulationContext


def build() -> SimulationContext:
    app = SimulationContext(sim_id="A5", factor=0, random_seed=42)
    app.add_arrival("src", "constant", value=1, off_mode="drop", spawn="t")
    app.add_arrival("switch_off", "constant", value=2.5)
    app.add_arrival("switch_on", "constant", value=5.5)
    app.add_arrival("src2", "constant", value=2, off_mode="restart", spawn="t")
    app.add_arrival("switch2_off", "constant", value=3)
    app.add_arrival("switch2_on", "constant", value=7)

    @app.task(name="t")
    def t(task_id):
        return {"event_type": "arrival"}

    @app.task(name="src_off", deactivate="src")
    def src_off(task_id):
        return {"event_type": "off"}

    @app.task(name="src_on", activate="src")
    def src_on(task_id):
        return {"event_type": "on"}

    @app.task(name="src2_off", deactivate="src2")
    def src2_off(task_id):
        return {"event_type": "off"}

    @app.task(name="src2_on", activate="src2")
    def src2_on(task_id):
        return {"event_type": "on"}

    @app.arrival_loop("switch_off")
    def switch_off_arrivals(context):
        for _ in range(1):
            entity = yield context.wait_for_arrival("switch_off")
            context.spawn(src_off(context.next_instance("src_off"), entity=entity))

    @app.arrival_loop("switch_on")
    def switch_on_arrivals(context):
        for _ in range(1):
            entity = yield context.wait_for_arrival("switch_on")
            context.spawn(src_on(context.next_instance("src_on"), entity=entity))

    @app.arrival_loop("switch2_off")
    def switch2_off_arrivals(context):
        for _ in range(1):
            entity = yield context.wait_for_arrival("switch2_off")
            context.spawn(src2_off(context.next_instance("src2_off"), entity=entity))

    @app.arrival_loop("switch2_on")
    def switch2_on_arrivals(context):
        for _ in range(1):
            entity = yield context.wait_for_arrival("switch2_on")
            context.spawn(src2_on(context.next_instance("src2_on"), entity=entity))

    return app


def run(app: SimulationContext):
    """Runs the model to the end its `run` section sets."""
    return app.run(until=12)


if __name__ == "__main__":
    from desflow import ConsoleEgress

    APP = build()
    APP.add_egress(ConsoleEgress(format="json"))
    run(APP)
